\begin{frame}{Klustringstyper}

    \begin{description}
        \item[Partionell:] Data är indelad i ett antal oöverlappande kluster.
        \item[Hierarkisk:] Delkluster är tillåtna, kluster är representerade som ett träd.  
    \end{description}
    \vspace{1cm}
    \begin{description}
        \item[Uteslutande:] Ett objekt tillhör ett kluster.
        \item[Överlappande:] Ett objekt hör till några kluster.
        \item[Fuzzy:] Ett objekt hör till olika kluster med en specifik sannolikhet.  
    \end{description}
    \vspace{1cm}

    \begin{description}
        \item[Fullständig:] Varje objekt är tillskrivet (minst) ett kluster.
        \item[Ofullständigt:] Vissa objekt är inte tillskrivna något kluster. 
    \end{description}
    
\end{frame}

\begin{frame}{Avstånds- och närhetsmått}
    Ett avstånds- eller närhetsmått beskriver hur lika eller olika
    två observationer är.

    \begin{itemize}
        \item \textbf{Avståndsmått:}
              Ett litet värde innebär att observationerna är lika.

        \item \textbf{Likhetsmått:}
              Ett stort värde innebär att observationerna är lika.

        \item Måttet beräknas utifrån observationernas variabelvärden.

        \item Valet av mått påverkar vilka observationer som betraktas
              som närliggande och därmed vilka kluster som bildas.
    \end{itemize}

    \begin{greenbox}
        Det finns inget avståndsmått som passar alla typer av data.
    \end{greenbox}
\end{frame}

\begin{frame}{Vanliga avstånds- och närhetsmått}
    \begin{itemize}
        \item \textbf{Euklidiskt avstånd:}
              Det vanliga raka avståndet mellan två punkter.

        \item \textbf{Manhattanavstånd:}
              Summan av skillnaderna för de olika variablerna.

        \item \textbf{Cosinuslikhet:}
              Jämför observationernas riktning snarare än deras storlek.

        \item \textbf{Jaccard-likhet:}
              Används ofta för asymmetriska binära data.

        \item \textbf{Gower-avstånd:}
              Kan användas för data med blandade variabeltyper.
    \end{itemize}

    \begin{greenbox}
        Numeriska, binära, kategoriska och blandade data kan kräva
        olika avstånds- eller närhetsmått.
    \end{greenbox}
\end{frame}

\begin{frame}{K-means: Val av startvärden}
    K-means behöver $K$ initiala centroider. Två klassiska metoder är:

    \begin{itemize}
        \item \textbf{Forgy-metoden:} Välj $K$ distinkta observationer
        slumpmässigt som initiala centroider.

        \item \textbf{Slumpmässig indelning:} Tilldela varje observation
        slumpmässigt till ett av de $K$ klustren och beräkna därefter
        centroiden för varje kluster.
    \end{itemize}

    \vspace{0.3cm}

    \begin{itemize}
        \item Forgy-metoden är vanligast i praktiken.
        \item \texttt{stats::kmeans(x, centers = K)} använder Forgy-metoden.
        \item Ofta används flera slumpmässiga starter, exempelvis
        \texttt{nstart = 50}.
    \end{itemize}
\end{frame}

\begin{frame}{K-means klustring: Detaljer}

    \begin{itemize}
        \item Låt $c_i$ vara centroid för kluster $i$. Låt $C_i$ vara en mängd med alla observationer i kluster $i$.
        \item Vi behöver ett avståndsmått
        \begin{itemize}
            \item Används för att mäta avståndet mellan $c_i$ och övriga observationer.
            \item Vanligast är euklidiskt avstånd,
            \begin{equation*}
                d(x,x') = \sqrt{\sum_{i=1}^{p}(x_i - x'_i)^2}
            \end{equation*}
            \item Finns andra val som kan göras.
        \end{itemize}
    \end{itemize}
    
\end{frame}



\begin{frame}{K-means klustring: Detaljer}

    \begin{itemize}
        \item K-means minimerar SSE
        \item SSE i ett kluster ges av
        \begin{equation*}
            E_{c_i} = \sum_{x \in C_i} d(x, c_i)^2.
        \end{equation*}
        \item Totala SSE för alla kluster
        \begin{equation*}
            \operatorname{SSE} = \sum_{i=1}^{K} E_{c_i} = \sum_{i=1}^{K} \sum_{x \in C_i} d(x, c_i)^2
        \end{equation*}
        \item I det euklidiska rummet beräknas centroider som
        \begin{equation*}
            c_i = \frac{1}{n_i} \sum_{x \in C_i} x.
        \end{equation*}
        \item K-means algoritmen hittar ett lokalt minima.
    \end{itemize}
    
\end{frame}

\begin{frame}{K-means klustring: Startvärden}

    Vi måste välja våra initiala gissningar för centroider. Detta val påverkar starkt utgången av algoritmen.

    Dåliga startvärden kan ge dåliga resultat.
    \includegraphics[width = \textwidth]{figs/dåliga start värden.png}

    Vanlig metod är att köra algoritmen många gånger med olika (slumpade) startvärden.
    
\end{frame}

\begin{frame}{Halverande K-means}

    \begin{itemize}
        \item Algoritm som motverkar problemet med startvärden.
        \item Dela upp datamängden i två kluster, välj ett och dela upp det i två osv.
        \begin{itemize}
            \item Valet av kluster kan göras med avseende på flest observation, störst SSE eller annat kriterie.
        \end{itemize}
        \item Uppdelningen kan liknas vid ett binärt träd.
    \end{itemize}
    
\end{frame}

\begin{frame}{Bisecting K-means}
    Bisecting K-means skapar klustren stegvis:

    \begin{enumerate}
        \item Börja med alla observationer i ett enda kluster.

        \item Välj ett befintligt kluster att dela.
        \begin{itemize}
            \item Ofta väljs klustret med störst SSE.
        \end{itemize}

        \item Dela det valda klustret genom att köra K-means
              med $K=2$.

        \item Ersätt det gamla klustret med de två nya klustren.

        \item Upprepa tills önskat antal kluster har skapats.
    \end{enumerate}

    \begin{greenbox}
        Till skillnad från vanlig K-means skapas inte alla $K$
        kluster samtidigt.
    \end{greenbox}
\end{frame}

\begin{frame}{K-means++}


    Algoritm för att hitta startvärden till K-means.

    \begin{enumerate}
        \item Välj en centroid uniformt slumpmässigt från observationerna.
        \item För varje datapunkt $x$ beräkna avstånder $d(x,c_i)$ mellan $x$ och den närmaste centroiden (som redan valts).
        \item Välj en datapunkt som centroid genom att:
        \begin{itemize}
            \item Slumpa en punkt med hjälp av viktade sannolikheter, där vikterna är proportionella mot $d(x,c_i)^2$, där $c_i$ är den närmaste centoriden för varje punkt.
        \end{itemize}
        \item Upprepa 2 och 3 tills $K$ centroider valts ut.
        \item Kör vanlig K-means med dessa startcentroider.
    \end{enumerate}
    
\end{frame}

\begin{frame}{K-means++}

    \begin{itemize}
        \item Generellt: K-means++ förbättrar SSE mycket över slumpade startvärden.
        \item Tar extra tid att bestämma startvärden, men K-means konvergerar mycket snabbare än med slumpade startvärden.
        \item Vanligt att K-means++ är dubbelt så snabb som K-means med slumpade startvärden.
    \end{itemize}
    
\end{frame}

\begin{frame}{K-means klustring: Kommentarer}

    \begin{itemize}
        \item Enkel och ganska effektiv.
        \item Känslig mot startvärden.
        \begin{itemize}
            \item Kör många gånger med olika startvärden.
            \item Halverande K-means.
            \item K-means++
        \end{itemize}
        \item Skapar klotformade kluster och är linjärt separerade.
        \begin{itemize}
            \item Om datas "naturliga kluster" har andra former fungerar k-means sämre.
        \end{itemize}
        \item Ger en centroid för varje kluster. Kan användas för att beskriva klustret.
        \item Har svårt att identifiera kluster av olika storlekar eller med olika tätheter.
        \item Känslig mot extremvärden.
    \end{itemize}
    
\end{frame}


\begin{frame}{Beräkningskomplexitet}
    Låt $n$ vara antal observationer, $p$ antal variabler,
    $K$ antal kluster och $I$ antal iterationer.

    \begin{itemize}
        \item \textbf{Vanlig K-means:}
        \[
            \mathcal{O}(npKI)
        \]

        \item \textbf{K-means++:}
        \[
            \mathcal{O}(npK) + \mathcal{O}(npKI)
        \]
        Dyrare initialisering, men ofta snabbare konvergens.

        \item \textbf{Bisecting K-means:}
        \[
          \mathcal{O}(npI\log K)
        \]
        Ofta snabbare än vanlig K-means när $K$ är stort.
    \end{itemize}

    \begin{greenbox}
        Antalet omstarter multiplicerar beräkningskostnaden.
    \end{greenbox}
\end{frame}

\begin{frame}{Utökningar av K-means}
    K-means kan generaliseras genom att byta närhetsmått och anpassa
    klusterrepresentanten.

    \begin{itemize}
        \item \textbf{Kvadrerat euklidiskt avstånd}
        \[
            d(x,c)^2 = \sum_{j=1}^{p}(x_j-c_j)^2
        \]
        Klusterrepresentant: komponentvisa \textbf{medelvärden}.

        \item \textbf{Manhattanavstånd ($L_1$)}
        \[
            d(x,c) = \sum_{j=1}^{p}|x_j-c_j|
        \]
        Klusterrepresentant: komponentvisa \textbf{medianer}
        (\textit{K-medians}).

        \item \textbf{Cosinusavstånd}
        \[
            d(x,c) = 1-\frac{x^\top c}{\lVert x\rVert\lVert c\rVert}
        \]
        Klusterrepresentant: \textbf{normaliserat medelvärde}
        (\textit{spherical K-means}).

    \end{itemize}
\end{frame}


\begin{frame}{K-medians}
    K-medians är en robust variant av K-means.

    \begin{itemize}
        \item Använder \textbf{Manhattanavstånd} ($L_1$):
        \[
            d(x,c_k)=\sum_{j=1}^{p}|x_j-c_{kj}|.
        \]

        \item Varje kluster representeras av den komponentvisa
              \textbf{medianen}:
        \[
            c_{kj}
            =
            \operatorname{median}\{x_{ij}:x_i\in C_k\}.
        \]

        \item Algoritmen upprepar två steg:
        \begin{enumerate}
           \item Tilldela varje observation till det närmaste klustercentrumet
             (med Manhattanavstånd).
            \item Uppdatera klustercentrumen genom att beräkna komponentvisa
              medianer inom varje kluster.
        \end{enumerate}

        \item Mindre känslig för extremvärden än vanlig K-means.

        \item Kan fortfarande konvergera till ett lokalt optimum och bör
              därför köras med flera startvärden.
    \end{itemize}
\end{frame}

\begin{frame}{K-means med cosinusavstånd}
    K-means med cosinusavstånd kallas ofta
    \textbf{spherical K-means}.

    \begin{itemize}
        \item Observationerna normaliseras till längd 1:
        \[
            \widetilde{x}_i
            =
            \frac{x_i}{\lVert x_i\rVert}.
        \]

        \item Klustertillhörighet bestäms med cosinusavstånd:
        \[
            d(x,c_k)
            =
            1-
            \frac{x^\top c_k}
                 {\lVert x\rVert\lVert c_k\rVert}.
        \]

        \item Varje observation tilldelas det klustercentrum som
              ger minst cosinusavstånd.

        \item Klustercentrumet beräknas som klustrets medelvärde
              och normaliseras därefter till längd 1.
    \end{itemize}
\end{frame}

\begin{frame}{Spherical K-means: Tillämpningar}
    Spherical K-means grupperar observationer efter
    \textbf{riktning} snarare än efter vektorernas storlek.

    \begin{itemize}
        \item \textbf{Dokumentklustring:}
              ordindikatorer, ordfrekvenser

        \item \textbf{Högdimensionella och glesa data:}
              många variabler men relativt få värden skilda från noll.

        \item \textbf{Vektorinbäddningar:}
              exempelvis representationer av text eller bilder.

        \item \textbf{Asymmetriska binära data:}
              när gemensamma ettor är informativa men gemensamma
              nollor inte ska påverka likheten.
    \end{itemize}

    \begin{greenbox}
        Metoden är särskilt användbar när vektorernas riktning är
        viktigare än deras längd.
    \end{greenbox}
\end{frame}

\begin{frame}{Utökningar av K-means}

    \begin{itemize}

        \end{itemize}
        \item K-medoids/Partitioning Around Medoids (PAM). Använder medioder som center (en punkt i datasetet).
    \end{itemize}
    
\end{frame}

\begin{frame}{Närhetsmått vid hierarkisk klustring}
    Vid agglomerativ hierarkisk klustring behöver vi göra två val:

    \begin{enumerate}
        \item \textbf{Avstånd mellan observationer}
        \begin{itemize}
            \item Euklidiskt avstånd
            \item Manhattanavstånd
            \item Cosinus- eller korrelationsavstånd
            \item Gower-avstånd för blandade datatyper
        \end{itemize}

        \item \textbf{Avstånd mellan kluster (länkningsmetod)}
        \begin{itemize}
            \item Single linkage
            \item Complete linkage
            \item Average linkage
            \item Ward's metod
        \end{itemize}
    \end{enumerate}

    \begin{greenbox}
        Både avståndsmåttet och länkningsmetoden kan påverka vilka
        kluster som skapas.
    \end{greenbox}
\end{frame}


\begin{frame}{Beräkning av avstånd mellan två kluster}

    Då kluster ofta innehåller flera observationer behövs en metod för att definera hur avstånd beräknas, även kallad \imp{länkningsmetod}.

    Låt $C_i$ och $C_j$ vara två kluster.
    \begin{itemize}
        \item MIN eller Single linkage (enkel länkning):
        \begin{equation*}
            \operatorname{prox}(C_i, C_j) = \min_{x \in C_i, y \in C_j} \operatorname{dist}(x, y).
        \end{equation*}
        \item MAX eller Complete linkage (fullständig länkning):
        \begin{equation*}
            \operatorname{prox}(C_i, C_j) = \max_{x \in C_i, y \in C_j} \operatorname{dist}(x, y).
        \end{equation*}
    \end{itemize}
    
\end{frame}

\begin{frame}{Beräkning av avständ mellan två kluster}

    \begin{itemize}
        \item Group average (genomsnitts länkning):
        \begin{equation*}
            \operatorname{prox}(C_i, C_j) = \frac{1}{n_i \cdot n_j} \sum_{x \in C_i, y \in C_j} \operatorname{dist}(x,y),
        \end{equation*}
        där $n_i$ och $n_j$ är antalet observationer i kluster $i$ och $j$.
        \item Wards/Centroid metod: Närheten defineras som hur mycket kvadrerade fel ökar när två kluster slås ihop.
        
        Samma kostnadsfunktion som i K-means.
    \end{itemize}
    
\end{frame}

\begin{frame}{Egenskaper}

    \begin{itemize}
        \item Ingen global funktion att optimera.
        \item Group average- och olika centroid metoder kan ta hänsyn till olika klusterstorlekar när ett par kluster förenas.
        \item Ihopslagningar är slutgiltiga och går inte att ta isär senare.
        \item Närhetsmåttet kan påverka resultatet.
        \begin{itemize}
            \item Extremvärden.
            \item Brus.
        \end{itemize}
        \item Passar bra för data som har en hierarkisk struktur.
    \end{itemize}
    
\end{frame}
